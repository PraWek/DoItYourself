# Простая но эффективная стратегия Гэндальфа

# Проверка здоровья и выбор тактики
(if (< (get-wizard-health "Гэндальф") 30)
    (progn
      (select-spell "Гэндальф" "ледяной осколок")
      (move-wizard "Гэндальф" (-1 -1)))
    (select-spell "Гэндальф" "огненный шар"))

# Движение к противнику если он виден
(if (is-wizard-visible "Гэндальф" "Мерлин")
    (progn
      (move-wizard "Гэндальф" (-1 0))
      (cast-spell "Гэндальф" "огненный шар" 15 10))
    (move-wizard "Гэндальф" (0 -1)))

# Адаптивное движение
(if (> (get-distance "Гэндальф" "Мерлин") 6)
    (move-wizard "Гэндальф" (-1 0))
    (if (< (get-distance "Гэндальф" "Мерлин") 3)
        (move-wizard "Гэндальф" (1 1))
        (select-spell "Гэндальф" "молния")))

# Атака при видимости противника
(if (is-wizard-visible "Гэндальф" "Мерлин")
    (cast-spell "Гэндальф" "молния" 14 8)
    (move-wizard "Гэндальф" (0 1)))

# Выбор заклинания по дистанции
(if (< (get-distance "Гэндальф" "Мерлин") 4)
    (select-spell "Гэндальф" "ледяной осколок")
    (select-spell "Гэндальф" "огненный шар"))

# Движение в сторону противника
(move-wizard "Гэндальф" (-1 0))

# Атака если противник близко
(if (< (get-distance "Гэндальф" "Мерлин") 5)
    (cast-spell "Гэндальф" "огненный шар" 13 9)
    (move-wizard "Гэндальф" (0 -1)))

# Защитная тактика при низком здоровье
(if (< (get-wizard-health "Гэндальф") 40)
    (progn
      (select-spell "Гэндальф" "ледяной осколок")
      (move-wizard "Гэндальф" (1 -1)))
    (move-wizard "Гэндальф" (-1 0)))

# Финальная атака
(if (is-wizard-visible "Гэндальф" "Мерлин")
    (progn
      (select-spell "Гэндальф" "молния")
      (cast-spell "Гэндальф" "молния" 12 7))
    (move-wizard "Гэндальф" (0 1)))

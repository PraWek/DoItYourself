# Простая но эффективная стратегия Мерлина

# Проверка здоровья и выбор защитной тактики
(if (< (get-wizard-health "Мерлин") 25)
    (progn
      (select-spell "Мерлин" "ледяной осколок")
      (move-wizard "Мерлин" (-1 -1)))
    (select-spell "Мерлин" "огненный шар"))

# Атака если Гэндальф виден
(if (is-wizard-visible "Мерлин" "Гэндальф")
    (progn
      (cast-spell "Мерлин" "огненный шар" 17 12)
      (move-wizard "Мерлин" (1 0)))
    (move-wizard "Мерлин" (1 1)))

# Движение в зависимости от дистанции
(if (> (get-distance "Мерлин" "Гэндальф") 7)
    (move-wizard "Мерлин" (1 1))
    (if (< (get-distance "Мерлин" "Гэндальф") 3)
        (move-wizard "Мерлин" (-1 -1))
        (select-spell "Мерлин" "молния")))

# Выбор заклинания по ситуации
(if (< (get-wizard-health "Мерлин") 50)
    (select-spell "Мерлин" "ледяной осколок")
    (if (> (get-distance "Мерлин" "Гэндальф") 5)
        (select-spell "Мерлин" "молния")
        (select-spell "Мерлин" "огненный шар")))

# Преследование противника
(if (is-wizard-visible "Мерлин" "Гэндальф")
    (move-wizard "Мерлин" (1 0))
    (move-wizard "Мерлин" (0 1)))

# Атака молнией на дальней дистанции
(if (> (get-distance "Мерлин" "Гэндальф") 6)
    (progn
      (select-spell "Мерлин" "молния")
      (cast-spell "Мерлин" "молния" 18 13))
    (move-wizard "Мерлин" (1 -1)))

# Тактическое позиционирование
(if (< (get-distance "Мерлин" "Гэндальф") 4)
    (cast-spell "Мерлин" "ледяной осколок" 18 12)
    (move-wizard "Мерлин" (0 1)))

# Агрессивная атака при хорошем здоровье
(if (> (get-wizard-health "Мерлин") 60)
    (progn
      (select-spell "Мерлин" "огненный шар")
      (move-wizard "Мерлин" (1 1)))
    (move-wizard "Мерлин" (0 -1)))

# Финальная комбинация
(if (is-wizard-visible "Мерлин" "Гэндальф")
    (progn
      (select-spell "Мерлин" "молния")
      (cast-spell "Мерлин" "молния" 19 14))
    (move-wizard "Мерлин" (1 0)))

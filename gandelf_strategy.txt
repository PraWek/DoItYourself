(let ((health (get-wizard-health "Гэндальф"))
      (enemy-visible (is-wizard-visible "Гэндальф" "Мерлин"))
      (distance (get-distance "Гэндальф" "Мерлин")))
  (if enemy-visible
      (if (< health 15)
          (move-wizard "Гэндальф" (-1 1))  ; Отступаем по диагонали
          (progn
              (select-spell "Гэндальф" "молния")
              (cast-spell "Гэндальф" "молния" 3 3)))
      ; Поисковое поведение - движение по спирали
      (let ((x (% (+ distance 1) 4)))
        (cond
          ((= x 0) (move-wizard "Гэндальф" (1 0)))
          ((= x 1) (move-wizard "Гэндальф" (0 1)))
          ((= x 2) (move-wizard "Гэндальф" (-1 0)))
          ((= x 3) (move-wizard "Гэндальф" (0 -1)))))))
